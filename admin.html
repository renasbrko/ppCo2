كود SQL لإنشاء الجدول في Supabase

CREATE TABLE files (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  title TEXT,
  notes TEXT,
  curriculum TEXT NOT NULL,
  subject TEXT NOT NULL,
  type TEXT NOT NULL,
  url TEXT NOT NULL,
  path TEXT,
  size_bytes BIGINT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- السماح بالقراءة العامة
ALTER TABLE files ENABLE ROW LEVEL SECURITY;
CREATE POLICY "authenticated read files" ON files FOR SELECT TO authenticated USING (true);
CREATE POLICY "admins insert files" ON files FOR INSERT TO authenticated WITH CHECK (public.is_admin());
CREATE POLICY "admins delete files" ON files FOR DELETE TO authenticated USING (public.is_admin());
